🧩 Conditional Logic Engine

approval routing (chain of roles)

Example:

export function evaluateRules(event, data, rules) {
  const matched = rules.filter(r => r.trigger === event);
  for (const rule of matched) {
    const pass = Object.entries(rule.conditions).every(([k, v]) => data[k] === v);
    if (pass) return rule.requiredFields;
  }
  return [];
}
This makes the workflow adaptive without writing new code per program.

🧾 PDF Generation Example (server action)
import PDFDocument from 'pdfkit';
import { uploadToStorage } from '@/lib/storage';

export async function generateExpensePDF(expense) {
  const doc = new PDFDocument();
  const chunks = [];
  doc.text(`Participant: ${expense.participantName}`);
  doc.text(`Amount: $${expense.amount}`);
  doc.image(expense.receiptPath, { width: 300 });
  doc.text(`Notes: ${expense.notes}`);
  doc.image(expense.signaturePath, { width: 150 });
  doc.end();

  for await (const chunk of doc) chunks.push(chunk);
  const pdfBuffer = Buffer.concat(chunks);
  const url = await uploadToStorage(pdfBuffer, `${expense.id}.pdf`);
  return url;
}
